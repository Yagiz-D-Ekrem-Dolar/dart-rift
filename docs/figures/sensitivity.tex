% Which observable sees which axis.
%
% Two observables, three interior axes, one bar each. The point is the
% shape, not the numbers: the two rows are almost disjoint. The momentum
% enhancement is an almost pure readout of matrix strength; the spin the
% impact adds is the only quantity here that responds to the boulder
% fraction -- the axis the momentum cannot see at all.
%
% Spearman rank correlation over the full 96-point production pool.
% Source: docs/rapor2204/veri/eksen_duyarliligi.dat, emitted by
% scripts/figure_data.py. No number is written here by hand.

% #1 row, #2 label, #3 r(beta), #4 r(omega)
\newcommand{\sens}[4]{%
  \pgfmathsetmacro{\yy}{-#1*\ROW}
  \node[dim, anchor=east, font=\small] at ({\BX-\HALF-5},\yy) {#2};
  % beta
  \fill[accent, opacity={abs(#3) > 0.5 ? 0.95 : 0.30}]
    (\BX,{\yy-\BARH}) rectangle ({\BX+#3*\HALF},{\yy+\BARH});
  % omega
  \fill[accent, opacity={abs(#4) > 0.3 ? 0.95 : 0.30}]
    (\WX,{\yy-\BARH}) rectangle ({\WX+#4*\HALF},{\yy+\BARH});
}
