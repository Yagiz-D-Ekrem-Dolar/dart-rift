% Where the five pilot runs sit in the cohesion--momentum landscape.
% Deliberately sparse: the earlier version carried a shaded prior box, a
% collapse arrow and six labels, and they overlapped each other and the
% fit line. Only annotations that carry a number survive.

\documentclass[border=3pt,tikz]{standalone}
\usepackage{pgfplots}
\usepackage{amsmath}
\usepackage[outline]{contour}
\usetikzlibrary{calc}
\pgfplotsset{compat=1.18}
\contourlength{1.1pt}
\tikzset{>=latex}

\colorlet{ink}{black!82}
\colorlet{sweepcol}{blue!55!cyan!78!black}
\colorlet{tenscol}{red!75!black}
\colorlet{offcol}{orange!90!black}
\colorlet{obscol}{green!45!black}
\colorlet{litcol}{violet!55!black}

\def\r{2.7pt}
\tikzset{run/.style={draw=none, ball color=#1, postaction={fill=#1,
  fill opacity=0.32, draw=#1!55!black, line width=0.13}}}

\begin{document}
\begin{tikzpicture}[font=\footnotesize]
% Least-squares power law through the three sweep points, refitted from
% the tabulated values: b = C Y0^p, C = 4.3511, p = -0.17905.
\pgfmathsetmacro{\C}{4.3511}
\pgfmathsetmacro{\p}{-0.17905}

\begin{axis}[
  width=76mm, height=54mm,
  xmode=log, ymode=log,
  xmin=4, xmax=1.4e4, ymin=0.055, ymax=5.2,
  xlabel={matrix cohesion $Y_0$\:[Pa]},
  ylabel={excess momentum $b=\beta-1$},
  xtick={10,100,1000,10000},
  xticklabels={$10$,$10^2$,$10^3$,$10^4$},
  ytick={0.1,1}, yticklabels={$0.1$,$1$},
  axis line style={draw=ink!65}, tick style={draw=ink!65},
  ticklabel style={scale=0.85, text=ink!80},
  label style={scale=0.92, text=ink},
  xlabel style={below=-2pt}, ylabel style={above=-4pt},
  axis on top=false,
]

  % --- published homogeneous range, for orientation only ---------------
  \fill[litcol!10] (axis cs:20,0.055) rectangle (axis cs:80,5.2);
  \node[scale=0.76, text=litcol, rotate=90, anchor=south] at (axis cs:32,0.30)
       {\contour{white}{Senel 20--80 Pa}};

  % --- mass-adjusted DART target ---------------------------------------
  \fill[obscol!14] (axis cs:4,2.2948) rectangle (axis cs:1.4e4,2.7298);
  \draw[densely dashed, obscol!80, line width=0.5pt]
       (axis cs:4,2.5418) -- (axis cs:1.4e4,2.5418);
  \node[scale=0.76, text=obscol!85!black, anchor=south east]
       at (axis cs:1.25e4,2.76) {DART target};

  % --- fitted power law -------------------------------------------------
  \addplot[domain=6:1.2e4, samples=50, sweepcol!60, line width=0.7pt,
           densely dotted, forget plot] {\C*x^\p};

  % --- the conditional interpolation ------------------------------------
  \draw[sweepcol!65, line width=0.5pt]
       (axis cs:20.13,0.055) -- (axis cs:20.13,2.5418);

  % --- the five completed runs ------------------------------------------
  \draw[run=sweepcol] (axis cs:10,2.7480) circle (\r);
  \draw[run=sweepcol] (axis cs:500,1.6247) circle (\r);
  \draw[run=sweepcol] (axis cs:5000,0.8738) circle (\r);
  \draw[run=offcol]   (axis cs:10,2.6650) circle (\r);
  \draw[run=tenscol]  (axis cs:10,0.0807) circle (\r);

  \node[scale=0.82, text=sweepcol!55!black, anchor=south west]
       at (axis cs:11.6,3.00) {\contour{white}{DY2}};
  \node[scale=0.82, text=sweepcol!55!black, anchor=north west]
       at (axis cs:560,1.50) {DO1};
  \node[scale=0.82, text=sweepcol!55!black, anchor=north west]
       at (axis cs:5600,0.81) {DO2};
  % DN sits almost on top of DY2 on a log axis; a short leader keeps
  % its label off both the marker and the observation band.
  \draw[offcol!65, line width=0.4pt]
       (axis cs:11.0,2.50) -- (axis cs:14.5,1.80);
  \node[scale=0.82, text=offcol!60!black, anchor=north west]
       at (axis cs:13.8,1.82) {DN, $+25$ m};
