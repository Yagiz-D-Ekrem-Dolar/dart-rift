\tikzset{
  etiket/.style={inner sep=1pt, font=\footnotesize},
  kilavuz/.style={line width=0.3pt, draw=#1,
                  shorten >=1.5pt, shorten <=1.5pt},
  altyazi/.style={anchor=north west, align=left, text=koyu!85,
                  inner sep=0pt, font=\footnotesize},
}

% Row filter: plot only the rows whose column #1 equals #2. PGFPlots has
% no built-in key for this; the idiom is to send unwanted rows to `inf`.
\pgfplotsset{
  discard if not/.style 2 args={
    x filter/.code={%
      \edef\satirdeger{\thisrow{#1}}%
      \edef\istenen{#2}%
      \ifx\satirdeger\istenen\else\def\pgfmathresult{inf}\fi
    }
  },
}

\pgfplotsset{
  temel/.style={
    width=56mm, height=50mm,
    scale only axis,
    axis line style={draw=koyu!60, line width=0.4pt},
    tick style={draw=koyu!60, line width=0.4pt},
    tick align=outside,
    every axis label/.style={font=\footnotesize, text=koyu},
    tick label style={font=\scriptsize, text=koyu!80},
    label style={font=\footnotesize},
    title style={font=\footnotesize, text=koyu, yshift=1pt},
    axis on top,
    % no grid: a grid line that carries no information is noise
  },
}
